\subsubsection{Decimale getallen en breuken}

\[0.4 + \dfrac{2}{3}+\dfrac{4}{5}= ?\]

Er zijn 2 mogelijkheden:
\begin{itemize}
  \item ofwel herleid je alle breuken naar decimale getallen;
  \item ofwel herleid je alle decimale getallen naar breuken.
\end{itemize}

\subsubsection*{1ste manier:}

\begin{voorbeeld}
\begin{align*}
  0.4 + \dfrac{2}{3} + \dfrac{4}{5} &= 0.4 + 0.666\ldots + 0.8\\
                                    &= 1.8666\ldots
\end{align*}
$0.666\ldots$ noemen we een oneindig repeterend decimaal getal. Het cijfer $6$, dat steeds terugkeert, noemen we de periode of het repeterend deel.
\end{voorbeeld}

\begin{voorbeeld}
\begin{align*}
  \dfrac{7}{22} &= 0.3181818\ldots
\end{align*}
De cijfers $1$ en $8$ worden steeds herhaald. Het repeterend deel is hier $18$.
Het repeterend deel wordt steeds minimum 3-maal geschreven.
Het cijfer $3$ wordt niet herhaald; dat is het niet-repeterend deel.
\end{voorbeeld}

\subsubsection*{2de manier:}
herleid alles naar breuk
\begin{align*}
  0.4 + \dfrac{2}{3} + \dfrac{4}{5} &= \dfrac{4}{10} + \dfrac{2}{3} + \dfrac{4}{5} & \text{herschrijf}\\
                                    &= \dfrac{2}{5} + \dfrac{2}{3} + \dfrac{4}{5} & \text{vereenvoudig}\\
                                    &= \dfrac{6}{15} + \dfrac{10}{15} + \dfrac{12}{15} & \text{maak gelijknamig}\\
                                    &= \dfrac{28}{15}\\
\end{align*}

\paragraph*{Opmerking:} Als er in de oefening repeterende decimale vormen voorkomen, werk dan bij voorkeur met breuken.
Zet breukstrepen, gelijkheidstekens en bewerkingstekens steeds op dezelfde horizontale lijn.

\subsection{Aftrekken van breuken}

Bij het aftrekken van breuken gebruiken we dezelfde werkwijze als bij het optellen, maar trekken de tellers van elkaar af.

\subsubsection{Gelijknamige breuken}

tellers aftrekken, noemer behouden.

\begin{voorbeeld}
$\dfrac{8}{13}-\dfrac{5}{13}=\dfrac{3}{13}$
\end{voorbeeld}


\subsubsection{Ongelijknamige breuken}

\begin{itemize}
  \item maak de breuken gelijknamig;
  \item maak het verschil van de tellers;
  \item indien mogelijk, vereenvoudig.
\end{itemize}

\subsection{Vermenigvuldigen van breuken}
\[\dfrac{3}{5}+\dfrac{3}{5}+\dfrac{3}{5}+\dfrac{3}{5}=4\times\dfrac{3}{5}=\dfrac{12}{5} \text{ en } 4=\dfrac{4}{1}\]
\[\text{dus } 4\times\dfrac{3}{5}=\dfrac{4}{1}\times\dfrac{3}{5}=\dfrac{12}{5}\]

\begin{onthoud}
Om een breuk te vermenigvuldigen met een breuk, vermenigvuldigen we teller met teller en noemer met noemer.
\end{onthoud}

\begin{voorbeeld}
$\dfrac{2}{3}\times\dfrac{5}{7}=\dfrac{2\times 5}{3\times 7}=\dfrac{10}{21}$
\end{voorbeeld}

\begin{voorbeeld}
$\dfrac{2}{9}\times\dfrac{8}{15}=\dfrac{2\times 8}{9\times 15}=\dfrac{16}{135}$
\end{voorbeeld}

\begin{voorbeeld}
\vspace*{-1cm}
\begin{align*}
  \dfrac{4}{9}\times\dfrac{45}{28} &= \dfrac{4\times 45}{9 \times 28} & \parbox{15em}{Vereenvoudig zo vlug mogelijk, pas hiervoor de hoofdeigenschap van de breuken toe.}\\
                                    &= \dfrac{1\times 45}{9 \times 7} & \parbox{15em}{Deel teller en noemer door 4.}\\
                                    &= \dfrac{1\times 5}{1 \times 7} & \parbox{15em}{Verder vereenvoudigen, deel teller en noemer door 9.}\\
                                    &= \dfrac{5}{7}\\
\end{align*}
Je hoeft niet alle tussenstappen te noteren, je mag de opgave ook onmiddellijk vereenvoudigen.
\end{voorbeeld}

\begin{voorbeeld}

\subsection{Delen van breuken}

De deling is de omgekeerde bewerking van de vermenigvuldiging.

\begin{onthoud}
Om te delen door een breuk, vermenigvuldig je met de omgekeerde breuk.
\end{onthoud}

\begin{voorbeeld}
$\dfrac{4}{3}:\dfrac{5}{8}=\dfrac{4}{3}\times\dfrac{8}{5}=\dfrac{32}{15}$
\end{voorbeeld}

\begin{voorbeeld}
$\dfrac{5}{8}:3=\dfrac{5}{8}\times\dfrac{1}{3}=\dfrac{5}{24}$
\end{voorbeeld}

\begin{voorbeeld}
$7:\dfrac{2}{5}=7\times\dfrac{5}{2}=\dfrac{35}{2}$
\end{voorbeeld}

\paragraph*{Opgelet!} Schrijf eerst als een vermenigvuldiging en probeer dan te vereenvoudigen. Vereenvoudig niet in een deling.

\begin{voorbeeld}
$\dfrac{8}{15}:\dfrac{3}{5}=\dfrac{8}{\underset{3}{\cancel{15}}}\times\dfrac{\cancel{5}}{3}=\dfrac{8}{9}$
\end{voorbeeld}

\subsubsection{Distributieve eigenschap}

\[\left(\dfrac{2}{5}+\dfrac{3}{8}\right)\times\dfrac{1}{7}\]
Deze oefening kan je, net als bij gehele getallen, op 2 manieren uitwerken.

\paragraph*{1ste manier:} werk eerst de haakjes uit
\[\left(\dfrac{2}{5}+\dfrac{3}{8}\right)\times\dfrac{1}{7}=\left(\dfrac{16}{40}+\dfrac{15}{40}\right)\times\dfrac{1}{7}=\dfrac{31}{40}\times\dfrac{1}{7}=\dfrac{31}{280}\]

\paragraph*{2de manier:} distributieve eigenschap
\[\left(\dfrac{2}{5}+\dfrac{3}{8}\right)\times\dfrac{1}{7}=\dfrac{2}{5}\times\dfrac{1}{7}+\dfrac{3}{8}\times\dfrac{1}{7}=\dfrac{2}{35}+\dfrac{3}{56}=\dfrac{16}{280}+\dfrac{15}{280}=\dfrac{31}{280}\]
\end{voorbeeld}

\begin{voorbeeld}
\[\left(\dfrac{15}{7}+\dfrac{13}{4}\right):\dfrac{1}{3}\]

\paragraph*{1ste manier:}
\[\left(\dfrac{60}{28}+\dfrac{91}{28}\right):\dfrac{1}{3}=\dfrac{151}{28}:\dfrac{1}{3}=\dfrac{151}{28}\times 3=\dfrac{453}{28}\]

\paragraph*{2de manier:}
\[\dfrac{15}{7}:\dfrac{1}{3}+\dfrac{13}{4}:\dfrac{1}{3}=\dfrac{15}{7}\times 3+\dfrac{13}{4}\times 3=\dfrac{45}{7}+\dfrac{39}{4}=\dfrac{180}{28}+\dfrac{273}{28}=\dfrac{453}{28}\]
\end{voorbeeld}

\subsection{Macht van een breuk}

\[\dfrac{3}{4}\times\dfrac{3}{4}\times\dfrac{3}{4}=\dfrac{3\times 3\times 3}{4\times 4\times 4}=\dfrac{27}{64}\quad\text{definitie vermenigvuldigen van breuken}\]
\[\dfrac{3}{4}\times\dfrac{3}{4}\times\dfrac{3}{4}=\left(\dfrac{3}{4}\right)^3=\dfrac{3^3}{4^3}=\dfrac{27}{64}\quad\text{definitie macht}\]

\begin{onthoud}
Om een breuk tot een macht te verheffen, verhef je teller en noemer tot die macht.
\end{onthoud}

\begin{voorbeeld}
$\left(\dfrac{2}{3}\right)^5=\dfrac{2^5}{3^5}=\dfrac{32}{243}$
\end{voorbeeld}

\begin{voorbeeld}
$\left(\dfrac{1}{7}\right)^2=\dfrac{1}{49}$
\end{voorbeeld}

\begin{voorbeeld}
$\left(\dfrac{1}{10}\right)^6=\dfrac{1}{\num{1000000}}$
\end{voorbeeld}

\begin{voorbeeld}
$\left(\dfrac{3}{11}\right)^0=1$\\
Een breuk tot de nulde macht is steeds 1.
\end{voorbeeld}

  \begin{enumerate}
    \item $\left(\dfrac{2}{7}\right)^3$
    \item $\left(\dfrac{4}{3}\right)^3$
    \item $\left(\dfrac{47}{9}\right)^2$
    \item $\left(\dfrac{93}{11}\right)^0$
  \end{enumerate}
\end{multicols}
\end{exercise}

\begin{solution}
  \begin{enumerate}

    \item $\left(\dfrac{4}{3}\right)^3=\dfrac{64}{27}$

    \item $\left(\dfrac{93}{11}\right)^0=1$
  \end{enumerate}
\end{solution}

\paragraph*{Belang van haakjes}
\[\left(\dfrac{2}{3}\right)^4\neq\dfrac{2^4}{3}\quad\text{want}\quad\left(\dfrac{2}{3}\right)^4=\dfrac{2^4}{3^4}=\dfrac{16}{81}\quad\text{en}\quad\dfrac{2^4}{3}=\dfrac{16}{3}\]

\subsection{Vierkantswortel van een breuk}

\[\sqrt{\dfrac{169}{196}}=\dfrac{13}{14}\quad\text{want}\quad\left(\dfrac{13}{14}\right)^2=\dfrac{169}{196}\quad\text{(zie definitie vierkantswortel les 5)}\]

\begin{onthoud}
Om een vierkantswortel te trekken uit een breuk, trek je de vierkantswortel uit de teller en uit de noemer van die breuk.
\end{onthoud}

\begin{voorbeeld}
$\sqrt{\dfrac{25}{49}}=\dfrac{5}{7}$ want $\left(\dfrac{5}{7}\right)^2=\dfrac{25}{49}$
\end{voorbeeld}

\begin{voorbeeld}
$\sqrt{\dfrac{36}{144}}=\dfrac{6}{12}=\dfrac{1}{2}$
\end{voorbeeld}

\begin{voorbeeld}
Grote getallen ontbind je eerst in priemfactoren.
\[\sqrt{\dfrac{1225}{1089}}=\sqrt{\dfrac{5^2\times 7^2}{3^2\times 11^2}}=\dfrac{5\times 7}{3\times 11}=\dfrac{35}{33}\]
\end{voorbeeld}

\begin{exercise}
Werk uit:
\begin{multicols}{2}
  \begin{enumerate}
    \item $\sqrt{\dfrac{1}{9}}$
    \item $\sqrt{\dfrac{121}{100}}$
    \item $\sqrt{\dfrac{18}{50}}$
    \item $\sqrt{\dfrac{49}{9}}$
  \end{enumerate}
\end{multicols}
\end{exercise}

\begin{solution}
  \begin{enumerate}
    \item $\sqrt{\dfrac{1}{9}}=\dfrac{1}{3}$
    \item $\sqrt{\dfrac{121}{100}}=\dfrac{11}{10}$
    \item $\sqrt{\dfrac{18}{50}}=\sqrt{\dfrac{9}{25}}=\dfrac{3}{5}$
    \item $\sqrt{\dfrac{49}{9}}=\dfrac{7}{3}$
  \end{enumerate}
\end{solution}

\begin{exercise}
Welke bewerking geeft het kleinste resultaat?
\begin{meerkeuze}
  \item $\dfrac{24}{5}\times\dfrac{2}{9}$
  \item $\dfrac{35}{6}:\dfrac{28}{9}$
  \item $\left(\dfrac{5}{4}\right)^2$
  \item $\dfrac{23}{6}-\dfrac{15}{8}$
\end{meerkeuze}
\end{exercise}

\begin{solution}
  \begin{enumerate}
    \item $\dfrac{24}{5}\times\dfrac{2}{9}=\dfrac{48}{45}=\dfrac{16}{15}$
    \item $\dfrac{35}{6}:\dfrac{28}{9}=\dfrac{35}{6}\times\dfrac{9}{28}=\dfrac{315}{168}=\dfrac{15}{8}$
    \item $\left(\dfrac{5}{4}\right)^2=\dfrac{25}{16}$
    \item $\dfrac{23}{6}-\dfrac{15}{8}=\dfrac{92}{24}-\dfrac{45}{24}=\dfrac{47}{24}$
  \end{enumerate}
Antwoord (A).
\end{solution}

\subsection{Volgorde van de bewerkingen}

De volgorde waarin de bewerkingen worden uitgevoerd wijzigt niet als er breuken in de oefeningen voorkomen.

\paragraph*{Overzicht (zie ook les 5)}
\begin{enumerate}[label=\arabic*.]
  \item Bewerkingen tussen haken (indien er binnen de haken nog haken voorkomen, werk ze dan uit van binnen naar buiten).
  \item Vierkantswortels en machten van links naar rechts.
  \item Vermenigvuldigen en delen van links naar rechts.
  \item Optellen en aftrekken van links naar rechts.
\end{enumerate}

\begin{voorbeeld}
\begin{align*}
  \dfrac{2}{3}-\dfrac{2}{11}:\dfrac{2}{7} &= \dfrac{2}{3}-\dfrac{2}{11}\times\dfrac{7}{2} & \parbox{15em}{eerst delen}\\
                                          &= \dfrac{2}{3}-\dfrac{7}{11} & \parbox{15em}{dan aftrekken}\\
                                          &= \dfrac{22}{33}-\dfrac{21}{33}=\dfrac{1}{33}
\end{align*}
\end{voorbeeld}

\begin{voorbeeld}
\begin{align*}
  \left[\dfrac{3}{8}+\left(\dfrac{3}{4}\right)^2\right]\times\dfrac{1}{3} &= \left[\dfrac{3}{8}+\dfrac{9}{16}\right]\times\dfrac{1}{3} & \parbox{15em}{binnen de haakjes, eerst machten}\\
                                                                          &= \left[\dfrac{6}{16}+\dfrac{9}{16}\right]\times\dfrac{1}{3} & \parbox{15em}{haken uitwerken}\\
                                                                          &= \dfrac{15}{16}\times\dfrac{1}{3}=\dfrac{5}{16} & \parbox{15em}{vermenigvuldigen (eerst vereenvoudigen)}
\end{align*}
Vereenvoudig steeds zo vlug mogelijk.
\end{voorbeeld}

\begin{voorbeeld}
\begin{align*}
  \dfrac{3}{13}+\left[\dfrac{2}{5}:\left(\dfrac{3}{8}\times\sqrt{\dfrac{4}{9}}\right)\right] &= \dfrac{3}{13}+\left[\dfrac{2}{5}:\left(\dfrac{3}{8}\times\dfrac{2}{3}\right)\right] & \parbox{15em}{binnen de haken, eerst vierkantswortel}\\
                                                                                              &= \dfrac{3}{13}+\left[\dfrac{2}{5}:\dfrac{1}{4}\right] & \parbox{15em}{ronde haken uitwerken}\\
                                                                                              &= \dfrac{3}{13}+\dfrac{8}{5} & \parbox{15em}{delen}\\
                                                                                              &= \dfrac{15}{65}+\dfrac{104}{65}=\dfrac{119}{65}
\end{align*}
Maak enkel gelijknamig bij optellen en aftrekken!
\end{voorbeeld}

\begin{exercise}
\begin{enumerate}
  \item $\left(\dfrac{3}{4}\right)^3\times\sqrt{\dfrac{256}{81}}$
  \item $\dfrac{3}{2}+\dfrac{10}{3}\times\dfrac{5}{4}$
  \item $\dfrac{3}{4}\times\dfrac{4}{5}+\dfrac{3}{8}\times\dfrac{7}{4}-\dfrac{11}{6}:\dfrac{5}{3}$
\end{enumerate}
\end{exercise}

\begin{solution}
  \begin{enumerate}
    \item $\left(\dfrac{3}{4}\right)^3\times\sqrt{\dfrac{256}{81}}=\dfrac{27}{64}\times\dfrac{16}{9}=\dfrac{3}{4}$
    \item $\dfrac{3}{2}+\dfrac{10}{3}\times\dfrac{5}{4}=\dfrac{3}{2}+\dfrac{25}{6}=\dfrac{9}{6}+\dfrac{25}{6}=\dfrac{34}{6}=\dfrac{17}{3}$
    \item $\dfrac{3}{4}\times\dfrac{4}{5}+\dfrac{3}{8}\times\dfrac{7}{4}-\dfrac{11}{6}:\dfrac{5}{3}=\dfrac{3}{5}+\dfrac{21}{32}-\dfrac{11}{10}=\dfrac{96}{160}+\dfrac{105}{160}-\dfrac{176}{160}=\dfrac{25}{160}=\dfrac{5}{32}$
  \end{enumerate}
\end{solution}
